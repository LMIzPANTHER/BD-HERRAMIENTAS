ENVIAR EL CATÁLOGO A LA PÁGINA WEB function doGet(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("HERRAMIENTAS"); const data = sheet.getDataRange().getValues(); const result = []; for (let i = 1; i < data.length; i++) { const row = data[i]; if (!row[0]) continue; result.push({ id: parseInt(row[0]), np: row[2] ? String(row[2]) : "", nz: row[3] ? String(row[3]) : "", tp: row[4] ? String(row[4]) : "", ds: row[5] ? String(row[5]) : "", gr: row[6] ? String(row[6]) : "", ub: row[7] ? String(row[7]) : "", ct: parseInt(row[8]) || 1 }); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // 2. RECIBIR NUEVAS HERRAMIENTAS DESDE LA PÁGINA WEB function doPost(e) { try { const payload = JSON.parse(e.postData.contents); if (payload.accion === 'ALTA_HERRAMIENTA') { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("HERRAMIENTAS"); sheet.appendRow([ payload.id, // A: ID "", // B: ID_FAMILIA payload.np, // C: NUM_PARTE_FAMILIA payload.nz, // D: NUM_PARTE_PIEZA payload.tp, // E: TIPO payload.ds, // F: DESCRIPCION payload.gr, // G: GRUPO payload.ub, // H: UBICACION payload.ct, // I: CANTIDAD TOTAL payload.ct, // J: CANTIDAD DISPONIBLE "DISPONIBLE", // K: ESTATUS "", // L: FOTO_URL "", // M: QR_UBICACION "", // N: NOTAS payload.fecha_alta // O: FECHA_ALTA ]); } return ContentService.createTextOutput(JSON.stringify({ status: 'ok' })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', msg: err.message })) .setMimeType(ContentService.MimeType.JSON); } }
